- [ ] **Step 3: Write minimal LineSystem**

XPBD (Macklin): `C = dist − L0`, `ã = α/Δt²`, `α = 1/k`. If `C < 0`, skip that constraint this iteration (unilateral). Otherwise `Δλ = (−C − ã λ) / (w + ã)` and apply the same rigid-body correction `RigidKiteSim.ProjectOne` already uses (`n = (attach−hand)/dist`, `w = 1/m + r×n · I⁻¹(r×n)`, `x += n λ/m`, rotate by `I⁻¹(r × nλ)`). Accumulate λ per line across 8 iterations. After the loop, `Tension = max(0, −λ / Δt²)`.

`rBody = 0` in these tests so attach is the CoM and both lines share the same constraint; that is enough to lock k and the slack branch.

Copy `InverseInertiaWorld` / `RotateWorld` into `LineSystem` as private methods (same bodies as `RigidKiteSim`). Do not make `RigidKiteSim`'s privates public.

```csharp
namespace Kitesim.Core
{
    public sealed class LineSystem
    {
        public const double RestLength = 45.0;
        public const double Stiffness = 750.0;
        public const int Iterations = 8;

        public double TensionLeft { get; private set; }
        public double TensionRight { get; private set; }

        public void Project(
            RigidKiteState state,
            FrameInertia inertia,
            Vec3 handLeft, Vec3 rLeftBody,
            Vec3 handRight, Vec3 rRightBody,
            double dt)
        {
            double lambdaL = 0.0;
            double lambdaR = 0.0;
            for (int i = 0; i < Iterations; i++)
            {
                ProjectOne(state, inertia, handLeft, rLeftBody, dt, ref lambdaL);
                ProjectOne(state, inertia, handRight, rRightBody, dt, ref lambdaR);
            }

            double scale = 1.0 / (dt * dt);
            TensionLeft = MathD.Max(0.0, -lambdaL * scale);
            TensionRight = MathD.Max(0.0, -lambdaR * scale);
        }

        private static void ProjectOne(
            RigidKiteState state,
            FrameInertia inertia,
            Vec3 hand,
            Vec3 rBody,
            double dt,
            ref double lambda)
        {
            Vec3 r = state.Orientation * rBody;
            Vec3 attach = state.Position + r;
            Vec3 delta = attach - hand;
            double dist = delta.Length();
            if (dist < 1e-12)
                return;

            Vec3 n = delta / dist;
            double c = dist - RestLength;
            if (c < 0.0)
            {
                lambda = 0.0;
                return;
            }

            Vec3 rxn = Vec3.Cross(r, n);
            Vec3 iinv = InverseInertiaWorld(inertia, state.Orientation, rxn);
            double w = 1.0 / inertia.Mass + Vec3.Dot(rxn, iinv);
            if (w < 1e-15)
                return;

            double atilde = (1.0 / Stiffness) / (dt * dt);
            double dLambda = (-c - atilde * lambda) / (w + atilde);
            lambda += dLambda;
            state.Position += n * (dLambda / inertia.Mass);
            Vec3 dOmegaWorld = InverseInertiaWorld(
                inertia, state.Orientation, Vec3.Cross(r, n * dLambda));
            state.Orientation = RotateWorld(state.Orientation, dOmegaWorld);
        }

        private static Vec3 InverseInertiaWorld(
            FrameInertia inertia, QuaternionD q, Vec3 torqueWorld)
        {
            Vec3 tBody = q.Conjugate() * torqueWorld;
            Vec3 aBody = new Vec3(
                tBody.X / inertia.Ixx,
                tBody.Y / inertia.Iyy,
                tBody.Z / inertia.Izz);
            return q * aBody;
        }

        private static QuaternionD RotateWorld(QuaternionD q, Vec3 rotVec)
        {
            double angle = rotVec.Length();
            if (angle < 1e-15)
                return q;
            return (QuaternionD.AxisAngle(rotVec, angle) * q).Normalized();
        }
    }
}
```

```csharp
namespace Kitesim.Core
{
    public static class PilotHands
    {
        public const double HalfWidth = 0.12;
        public const double Travel = 0.7;

        public static void Apply(RigidKiteSim sim, Vec3 stance, double left, double right)
        {
            double l = MathD.Clamp(left, -1.0, 1.0);
            double r = MathD.Clamp(right, -1.0, 1.0);
            sim.HandLeft = stance + new Vec3(-HalfWidth, -l * Travel, 0.0);
            sim.HandRight = stance + new Vec3(HalfWidth, -r * Travel, 0.0);
        }
    }
}
```

```csharp
[Test]
public void TurnRateScalesWithHandDifferenceAndWithWindSpeed()
{
    double r06v6 = MeanTurnRate(wind: 6.0, rightAxis: 0.6);
    double r03v6 = MeanTurnRate(wind: 6.0, rightAxis: 0.3);
    double r06v8 = MeanTurnRate(wind: 8.0, rightAxis: 0.6);

    Assert.Greater(r03v6, 0.02);
    Assert.AreEqual(2.0, r06v6 / r03v6, 0.8, "rate should roughly double with u");
    Assert.AreEqual(8.0 / 6.0, r06v8 / r06v6, 0.55, "rate should roughly scale with v_app");
}

private static double MeanTurnRate(double wind, double rightAxis)
{
    RigidKiteSim sim = RigidKiteSim.HundredSixteen();
    sim.Wind.Speed = wind;
    RigidKiteState state = sim.SpawnDownwind(elevationY: 15.0);
    for (double t = 0.0; t < 2.0; t += 1.0 / 50.0)
        sim.Step(state, 1.0 / 50.0);

    PilotHands.Apply(sim, Vec3.Zero, 0.0, rightAxis);
    double h0 = sim.View(state).HeadingRad;
    for (double t = 0.0; t < 1.0; t += 1.0 / 50.0)
        sim.Step(state, 1.0 / 50.0);
    return sim.View(state).HeadingRad - h0;
}
```
